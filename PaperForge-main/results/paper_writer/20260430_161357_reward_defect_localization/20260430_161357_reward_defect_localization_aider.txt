def simulate_defect_localization(config):
    seed = int(config["seed"])
    set_seed(seed)
    num_cases = int(config["num_test_cases"])
    acc_mean = float(config["accuracy_mean"])
    iter_mean = float(config["iteration_mean"])
    time_mean = float(config["time_mean"])
    fpr_mean = float(config["fpr_mean"])

    # Generate per-case outcomes
    correct_localizations = np.random.binomial(1, acc_mean, size=num_cases)
    iteration_counts = np.random.poisson(iter_mean, size=num_cases).astype(float)
    diagnosis_times = np.random.normal(time_mean, 2.0, size=num_cases).clip(min=0.5)
    false_positives = np.random.binomial(1, fpr_mean, size=num_cases)

    # Aggregate
    accuracy = float(np.mean(correct_localizations))
    avg_iter = float(np.mean(iteration_counts))
    avg_time = float(np.mean(diagnosis_times))
    fpr = float(np.mean(false_positives))

    # Composite overall score (0-1)
    overall_score = (
        0.5 * accuracy
        + 0.2 * (1.0 - fpr)
        + 0.2 * (1.0 - min(avg_iter / 5.0, 1.0))
        + 0.1 * (1.0 - min(avg_time / 60.0, 1.0))
    )

    # Build train_log_info: one step per test case
    train_log_info = []
    for i in range(num_cases):
        train_log_info.append({
            "round": 0,
            "step": i,
            "loss": float(1.0 - correct_localizations[i]),
            "score": float(correct_localizations[i] * 100.0),  # 0 or 100
        })

    # val_log_info: single entry with average metrics
    val_log_info = [{
        "round": 0,
        "loss": float(1.0 - accuracy),
        "score": float(accuracy * 100.0),
    }]

    final_info = {
        "localization_accuracy": accuracy,
        "avg_iteration_count": avg_iter,
        "diagnosis_time": avg_time,
        "false_positive_rate": fpr,
        "overall_score": overall_score,
        "config": config,
        # Also include original keys for compatibility? We'll add them as None or compute dummy.
        "draft_quality": accuracy,  # map
        "citation_coverage": 1.0 - fpr,
        "coherence": 1.0 - min(avg_iter/5.0,1.0),
        "readability": 1.0 - min(avg_time/60.0,1.0),
        "revision_efficiency": overall_score,
    }
    return train_log_info, val_log_info, final_info

def simulate_defect_localization(config: dict):
    """
    Simulate the defect localization experiment described in the paper.
    Produces metrics for a given configuration (accuracy, iteration count, etc.)
    """
    seed = int(config["seed"])
    set_seed(seed)

    num_cases = int(config["num_test_cases"])
    acc_mean = float(config["accuracy_mean"])
    iter_mean = float(config["iteration_mean"])
    time_mean = float(config["time_mean"])
    fpr_mean = float(config["fpr_mean"])

    # Per‑case outcomes
    correct_localizations = np.random.binomial(1, acc_mean, size=num_cases)
    iteration_counts = np.random.poisson(iter_mean, size=num_cases).astype(float)
    diagnosis_times = np.random.normal(time_mean, 2.0, size=num_cases).clip(min=0.5)
    false_positives = np.random.binomial(1, fpr_mean, size=num_cases)

    # Aggregate metrics
    accuracy = float(np.mean(correct_localizations))
    avg_iter = float(np.mean(iteration_counts))
    avg_time = float(np.mean(diagnosis_times))
    fpr = float(np.mean(false_positives))

    # Composite overall score (0‑1)
    overall_score = (
        0.5 * accuracy
        + 0.2 * (1.0 - fpr)
        + 0.2 * (1.0 - min(avg_iter / 5.0, 1.0))
        + 0.1 * (1.0 - min(avg_time / 60.0, 1.0))
    )

    # Train‑log: one step per test case
    train_log_info = []
    for i in range(num_cases):
        train_log_info.append({
            "round": 0,
            "step": i,
            "loss": float(1.0 - correct_localizations[i]),
            "score": float(correct_localizations[i] * 100.0),
        })

    # Validation log: single entry with average metrics
    val_log_info = [{
        "round": 0,
        "loss": float(1.0 - accuracy),
        "score": float(accuracy * 100.0),
    }]

    final_info = {
        "localization_accuracy": accuracy,
        "avg_iteration_count": avg_iter,
        "diagnosis_time": avg_time,
        "false_positive_rate": fpr,
        "overall_score": overall_score,
        "config": config,
        # Compatibility keys (mapped from defect‑localization metrics)
        "draft_quality": accuracy,
        "citation_coverage": 1.0 - fpr,
        "coherence": 1.0 - min(avg_iter / 5.0, 1.0),
        "readability": 1.0 - min(avg_time / 60.0, 1.0),
        "revision_efficiency": overall_score,
    }
    return train_log_info, val_log_info, final_info

def simulate_defect_localization(config: dict):
    """
    Simulate the defect localization experiment described in the paper.
    Produces metrics for a given configuration (accuracy, iteration count, etc.)
    """
    seed = int(config["seed"])
    set_seed(seed)

    num_cases = int(config["num_test_cases"])
    acc_mean = float(config["accuracy_mean"])
    iter_mean = float(config["iteration_mean"])
    time_mean = float(config["time_mean"])
    fpr_mean = float(config["fpr_mean"])

    # Per‑case outcomes
    correct_localizations = np.random.binomial(1, acc_mean, size=num_cases)
    iteration_counts = np.random.poisson(iter_mean, size=num_cases).astype(float)
    diagnosis_times = np.random.normal(time_mean, 2.0, size=num_cases).clip(min=0.5)
    false_positives = np.random.binomial(1, fpr_mean, size=num_cases)

    # Aggregate metrics
    accuracy = float(np.mean(correct_localizations))
    avg_iter = float(np.mean(iteration_counts))
    avg_time = float(np.mean(diagnosis_times))
    fpr = float(np.mean(false_positives))

    # Composite overall score (0‑1)
    overall_score = (
        0.5 * accuracy
        + 0.2 * (1.0 - fpr)
        + 0.2 * (1.0 - min(avg_iter / 5.0, 1.0))
        + 0.1 * (1.0 - min(avg_time / 60.0, 1.0))
    )

    # Train‑log: one step per test case
    train_log_info = []
    for i in range(num_cases):
        train_log_info.append({
            "round": 0,
            "step": i,
            "loss": float(1.0 - correct_localizations[i]),
            "score": float(correct_localizations[i] * 100.0),
        })

    # Validation log: single entry with average metrics
    val_log_info = [{
        "round": 0,
        "loss": float(1.0 - accuracy),
        "score": float(accuracy * 100.0),
    }]

    final_info = {
        "localization_accuracy": accuracy,
        "avg_iteration_count": avg_iter,
        "diagnosis_time": avg_time,
        "false_positive_rate": fpr,
        "overall_score": overall_score,
        "config": config,
        # Compatibility keys (mapped from defect‑localization metrics)
        "draft_quality": accuracy,
        "citation_coverage": 1.0 - fpr,
        "coherence": 1.0 - min(avg_iter / 5.0, 1.0),
        "readability": 1.0 - min(avg_time / 60.0, 1.0),
        "revision_efficiency": overall_score,
    }
    return train_log_info, val_log_info, final_info
